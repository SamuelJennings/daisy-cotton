{# @description One tab of a row of tabs, used inside tabs. Renders as a link when an href is given and as a button otherwise. #} {# @prop text:text | description:"The tab's label." #} {# @prop href:text | description:"Link address. Without it the tab is a button." #} {# @prop active:boolean | description:"Mark this tab as the current one." #} {# @prop disabled:boolean | description:"Show the tab as unavailable. A link loses its href and a button is disabled." #} {# @prop class:text | description:"Extra classes added to the tab." #} {# @slot — Extra content shown after the text. #} {% if href %} {{ text }}{{ slot }} {% else %} {% endif %}
